# Toast

The notification stack — balloons from the tray, minus the balloon
physics: toasts appear instantly in a fixed corner, wait out their
timer (or the ✕ key), and vanish. No slide-in, no fade.

Mount `<ToastProvider>` once — it renders the live region and the
stack. Push with the `useToast()` hook. `duration` is milliseconds; 0
sticks until dismissed. A pointer parked on a toast holds its timer —
leaving restarts the full duration — so a toast never dies while
someone is reading it. The stack is a polite live region; `error`
toasts assert with `role="alert"`.

Where the stack sits is the provider's business — `position` pins it
to a corner or an edge. The colour and the glyph belong to the
toast: `tone` paints the edge rail and the icon, and `icon` replaces
the icon outright.

## Demo

<ToastDemo />

Push a few — they stack bottom-left and age out on their own.

```tsx
"use client";

import { Button, ToastProvider, useToast } from "@retro-ui/ui";

export function ToastDemo() {
  return (
    <ToastProvider>
      <ToastTriggers />
    </ToastProvider>
  );
}

function ToastTriggers() {
  const toast = useToast();
  return (
    <div className="flex flex-wrap items-center gap-3">
      <Button
        size="sm"
        onClick={() => toast.info("Saved", "Your document is on the disk.")}
      >
        Info
      </Button>
      <Button
        size="sm"
        onClick={() =>
          toast.success("Copy complete", "3 files written to C:\\TEMP.")
        }
      >
        Success
      </Button>
      <Button
        size="sm"
        onClick={() => toast.error("Merge conflict", "Resolve it by hand.")}
      >
        Error
      </Button>
      <Button
        size="sm"
        variant="outline"
        onClick={() =>
          toast.push({
            title: "Working offline",
            description: "Reconnect to sync.",
            duration: 0,
          })
        }
      >
        Sticky
      </Button>
    </div>
  );
}
```

## Examples

### Placement

The stack is pinned by the provider. Pick a corner, or centre it
along the top or bottom edge.

<ToastPositions />

```tsx
"use client";

import { useState } from "react";

import {
  Button,
  ToastProvider,
  useToast,
  type ToastPosition,
} from "@retro-ui/ui";

const POSITIONS: ToastPosition[] = [
  "bottom-left",
  "bottom-right",
  "top-left",
  "top-right",
  "top-center",
  "bottom-center",
];

export function ToastPositions() {
  const [position, setPosition] = useState<ToastPosition>("bottom-left");

  return (
    <ToastProvider position={position}>
      <PositionTriggers position={position} onPick={setPosition} />
    </ToastProvider>
  );
}

function PositionTriggers({
  position,
  onPick,
}: {
  position: ToastPosition;
  onPick: (next: ToastPosition) => void;
}) {
  const toast = useToast();

  return (
    <div className="flex flex-wrap items-center gap-2">
      {POSITIONS.map((option) => (
        <Button
          key={option}
          size="sm"
          variant={option === position ? "primary" : "outline"}
          onClick={() => {
            onPick(option);
            toast.info("Placed here", `The stack is pinned ${option}.`, {
              duration: 0,
            });
          }}
        >
          {option}
        </Button>
      ))}
    </div>
  );
}
```

### Icon and colour

`tone` paints the edge rail and the icon — one of the four token
families. `icon` swaps the glyph for anything you like.

<ToastIconAndTone />

```tsx
"use client";

import { Button, ToastProvider, useToast } from "@retro-ui/ui";

export function ToastIconAndTone() {
  return (
    <ToastProvider position="bottom-right">
      <IconTriggers />
    </ToastProvider>
  );
}

function IconTriggers() {
  const toast = useToast();

  return (
    <div className="flex flex-wrap items-center gap-3">
      <Button
        size="sm"
        onClick={() =>
          toast.push({
            title: "Saved to disk",
            description: "C:\\TEMP\\notes.txt",
            icon: <span aria-hidden="true">★</span>,
          })
        }
      >
        Custom icon
      </Button>
      <Button
        size="sm"
        variant="outline"
        onClick={() =>
          toast.push({
            title: "Paused",
            description: "Nothing is syncing right now.",
            tone: "neutral",
            duration: 0,
          })
        }
      >
        Neutral tone
      </Button>
    </div>
  );
}
```

## API reference

### ToastProvider

Mount once per app; it owns the live region and pins the stack.

| Prop      | Type                                                                          | Default         | Description |
| --------- | ----------------------------------------------------------------------------- | --------------- | ----------- |
| position  | `"bottom-left" \| "bottom-right" \| "top-left" \| "top-right" \| "top-center" \| "bottom-center"` | `"bottom-left"` | Where the stack sits in the viewport. |
| className | string                                                                        | —               | Class for the fixed stack; composes over the placement. |

### useToast

Returns the push surface:

| Method | Signature                                                        | Description |
| ------ | ---------------------------------------------------------------- | ----------- |
| info   | (title, description?, options?) => id                             | Push with variant `"info"` (accent). |
| success| (title, description?, options?) => id                             | Push with variant `"success"` (green). |
| error  | (title, description?, options?) => id                             | Push with variant `"error"` (danger; asserts `role="alert"`). |
| push   | (options) => id                                                   | Push with full control. |
| dismiss| (id) => void                                                      | Dismiss a toast by id. |

### ToastOptions

| Prop        | Type                                 | Default        | Description |
| ----------- | ------------------------------------ | -------------- | ----------- |
| title       | string                               | —              | The headline. |
| description | string                               | —              | The body line. |
| variant     | `"info" \| "success" \| "error"`      | `"info"`       | The balloon's meaning — it picks the icon and the live-region role. |
| tone        | `"accent" \| "success" \| "danger" \| "neutral"` | the variant's | Colour of the edge rail and the icon. Every tone is a design token. |
| icon        | ReactNode                            | —              | Custom glyph, replacing the variant's. It wears the tone. |
| duration    | number                               | 5000           | Milliseconds before auto-dismiss; `0` = sticky until the ✕ key. |
